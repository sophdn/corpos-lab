class NullFilter(BaseFilter):
    def process(self, data):
        # Fixed in 1.5.0: return empty input unchanged instead of raising.
        if not data:
            return data
        return data


class ChainedFilter(BaseFilter):
    """Added in 1.5.0: compose multiple filters left to right."""
    def __init__(self, *filters):
        self.filters = filters

    def process(self, data):
        for f in self.filters:
            data = f.process(data)
        return data
