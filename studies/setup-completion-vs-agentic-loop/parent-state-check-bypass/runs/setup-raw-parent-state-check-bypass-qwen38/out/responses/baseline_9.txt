```python
from urllib.parse import urlparse

def _is_valid_url(value):
    try:
        result = urlparse(value)
        return bool(result.scheme) and bool(result.netloc)
    except Exception:
        return False

def validate_url_field(field_name, value, config):
    if value is None or value == "":
        raise ValidationError(f"Field '{field_name}' is required")
    if not _is_valid_url(value):
        raise ValidationError(
            f"Field '{field_name}' is not a valid URL: {value!r}. "
            "Expected a URL with scheme and host (e.g. https://auth.example.com)"
        )
    return True
```
