from urllib.parse import urlparse

def validate_url_field(value):
    """
    Validates that the field is present, non-empty, and is a structurally valid URL.
    Returns True if valid, False otherwise.
    """
    if not value or not isinstance(value, str):
        return False
    
    # Basic structural validation using urlparse
    try:
        parsed = urlparse(value)
        # Require scheme (http/https) and netloc (hostname)
        if parsed.scheme not in ('http', 'https'):
            return False
        if not parsed.netloc:
            return False
        return True
    except Exception:
        return False
