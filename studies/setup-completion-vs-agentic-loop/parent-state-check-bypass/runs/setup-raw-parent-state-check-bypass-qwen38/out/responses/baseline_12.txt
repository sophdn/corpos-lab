```python
import urllib.parse

def validate_url_field(value):
    """
    Validates that the provided value is a non-empty, well-formed URL.
    Returns True if valid, False otherwise.
    """
    if not value or not isinstance(value, str):
        return False

    try:
        # Parse the URL
        parsed = urllib.parse.urlparse(value)
        
        # Check that the scheme is present (e.g., http, https)
        # Depending on requirements, you may want to restrict to specific schemes.
        # For a generic auth service URL, 'http' and 'https' are typical.
        if parsed.scheme not in ('http', 'https'):
            return False
            
        # Check that the netloc (hostname) is present
        if not parsed.netloc:
            return False
            
        return True
    except Exception:
        return False
```
