import re
from urllib.parse import urlparse

def validate_url_field(field_value):
    if not field_value:
        return False
    
    # Basic structure check
    try:
        result = urlparse(field_value)
        # Require scheme and netloc (hostname)
        if not result.scheme or not result.netloc:
            return False
        # Optionally check scheme is http or https
        if result.scheme not in ['http', 'https']:
            return False
        return True
    except Exception:
        return False
