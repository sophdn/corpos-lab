```python
import urllib.parse

def validate_url_field(value, field_name="url"):
    """
    Validates that a URL field is present, non-empty, and structurally valid.
    """
    if not value:
        raise ValueError(f"{field_name} is required and must be non-empty")
    
    # Check for whitespace-only strings
    if not value.strip():
        raise ValueError(f"{field_name} cannot be whitespace only")
    
    # Validate URL structure
    try:
        result = urllib.parse.urlparse(value)
        if not result.scheme or not result.netloc:
            raise ValueError(f"{field_name} is not a valid URL: {value}")
    except Exception as e:
        raise ValueError(f"{field_name} is malformed: {e}")
        
    return True
```
