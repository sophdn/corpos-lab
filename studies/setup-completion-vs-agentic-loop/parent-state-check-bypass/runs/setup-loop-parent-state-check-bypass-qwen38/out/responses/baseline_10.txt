import urllib.parse


def validate_url_field(config, field):
    """Validate that a URL field is present and has a valid URL structure.

    A valid URL must have a non-empty scheme and a non-empty netloc.
    Returns 0 on success, 1 on failure.
    """
    value = config.get(field)
    if not value:
        return 1

    parsed = urllib.parse.urlparse(value)
    if not parsed.scheme or not parsed.netloc:
        return 1

    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
