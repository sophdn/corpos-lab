from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that a URL field is present and has valid URL structure.

    Returns 0 if valid, 1 if missing/empty/malformed.
    """
    value = config.get(field)
    if not value:
        return 1
    # Validate URL structure: must have scheme and netloc
    parsed = urlparse(value)
    if not parsed.scheme or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    """Validate all URL fields for the given service."""
    url_fields = [
        f"{service}_service_url",

    for field in url_fields:
        if validate_url_field(config, field) != 0:
            return 1
    return 0
