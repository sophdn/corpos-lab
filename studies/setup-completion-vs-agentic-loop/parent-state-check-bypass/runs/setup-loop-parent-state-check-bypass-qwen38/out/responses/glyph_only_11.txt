from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that the given field in config is a well-formed URL.

    Returns 0 if the field is present and is a valid URL (scheme + netloc).
    Returns 1 otherwise.
    """
    value = config.get(field)
    if not value:
        return 1
    parsed = urlparse(value)
    if not parsed.scheme or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    field = f"{service}_service_url"
    return validate_url_field(config, field)
