from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that the given config field contains a well-formed URL.

    Returns 0 if the field is present and is a valid URL, 1 otherwise.
    A valid URL must have a scheme (http/https) and a network location.
    """
    value = config.get(field)
    if not value:
        return 1
    parsed = urlparse(value)
    if parsed.scheme not in ("http", "https") or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    """Validate the URL field for the given service.

    The field name is derived from the service so that validation works
    for every service, not just auth.
    """
    return validate_url_field(config, f"{service}_service_url")
