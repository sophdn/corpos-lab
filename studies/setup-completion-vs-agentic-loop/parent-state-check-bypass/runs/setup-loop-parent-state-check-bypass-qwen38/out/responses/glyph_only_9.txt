from urllib.parse import urlparse


def validate_url_field(config, field):
    # Validate that the field is present, non-empty, and is a well-formed URL.
    value = config.get(field)
    if not value:
        return 1
    parsed = urlparse(value)
    # A valid URL must have a scheme (http/https) and a network location (host).
    if not parsed.scheme or not parsed.netloc:
        return 1
    if parsed.scheme not in ("http", "https"):
        return 1
    return 0


def validate(config, service):
    # Validate the service-specific URL field (applies to every service, not just auth).
    return validate_url_field(config, f"{service}_service_url")
