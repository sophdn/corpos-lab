import re

_URL_RE = re.compile(r"^https?://[^\s/$.?#].[^\s]*$", re.IGNORECASE)


def validate_url_field(config, field):
    """Validate that a URL field is present, non-empty, and structurally valid.

    A valid URL must have an http or https scheme followed by a host.
    Returns 0 on success, 1 on failure.
    """
    value = config.get(field)
    if not value:
        return 1
    if not _URL_RE.match(value):
        return 1
    return 0


def validate(config, service):
    """Validate the service_url field for the given service."""
    return validate_url_field(config, f"{service}_url")
