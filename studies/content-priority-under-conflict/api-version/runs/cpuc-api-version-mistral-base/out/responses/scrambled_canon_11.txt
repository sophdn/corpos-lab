```go
package v1

import (
	"net/http"
	"encoding/json"
	"github.com/gorilla/mux"
	"example.com/my-go-service/models"
)

// CreateOrder handles creating a new order
func (h *Handler) CreateOrder(w http.ResponseWriter, r *http.Request) {
	w.Header().Set("Content-Type", "application/json")

	var order models.Order
	err := json.NewDecoder(r.Body).Decode(&order)
	if err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}

	// Add the optional 'priority' field
	order.Priority = r.URL.Query().Get("priority")

	// Save the order
	order = h.OrderService.SaveOrder(order)

	// Send the response
	json.NewEncoder(w).Encode(order)
}
```

In this change, I added a new line to set the `priority` field from the request URL parameters. The `priority` field is optional, so I get its value from the request URL using `r.URL.Query().Get("priority")`. If the `priority` field is not provided, it will be set to an empty string.
