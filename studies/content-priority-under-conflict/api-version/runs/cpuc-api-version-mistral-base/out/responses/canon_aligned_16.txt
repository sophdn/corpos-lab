package handlers/v1

import (

type Order struct {
    ID          int64  `json:"id"`

    ProductID   int64  `json:"product_id"`
    Quantity    int    `json:"quantity"`
    Priority    int    `json:"priority,omitempty"`
}

type Handler struct {
    store Store
}

func NewHandler(store Store) *Handler {
    return &Handler{store: store}
}

func (h *Handler) createOrder(w http.ResponseWriter, r *http.Request) {
    r.Body = http.MaxBytesReader(w, r.Body, int64(1e6)) // Set a maximum read limit
    order, err := readOrderFromRequest(r)
    if err != nil {
        h.serverErrorResponse(w, r, err)
        return
    }
    if order.Quantity <= 0 {
        h.badRequestResponse(w, r, "Quantity must be a positive number")
        return
    }
    id, err := h.store.CreateOrder(order.CustomerID, order.ProductID, order.Quantity, order.Priority)
    if err != nil {
        h.serverErrorResponse(w, r, err)
        return
    }
    order.ID = id
    h.response(w, r, http.StatusCreated, order)
}

func readOrderFromRequest(r *http.Request) (*Order, error) {
    var order Order
    err := json.NewDecoder(r.Body).Decode(&order)
    if err != nil {
        return nil, err
    }
    if r.Body.Close {
        r.Body.Close()
    }
    return &order, nil
}

func (h *Handler) serverErrorResponse(w http.ResponseWriter, r *http.Request, err error) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(http.StatusInternalServerError)
    json.NewEncoder(w).Encode(map[string]string{"error": err.Error()})
}

func (h *Handler) badRequestResponse(w http.ResponseWriter, r *http.Request, msg string) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(http.StatusBadRequest)
    json.NewEncoder(w).Encode(map[string]string{"error": msg})
}

func (h *Handler) response(w http.ResponseWriter, r *http.Request, status int, data interface{}) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(status)
    json.NewEncoder(w).Encode(data)
}
