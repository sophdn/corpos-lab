// handlers/v1/orders.go

package handlers

import (
	"net/http"
	"encoding/json"
	"github.com/gorilla/mux"
	"example.com/myservice/api/models"
)

func CreateOrder(w http.ResponseWriter, r *http.Request) {
	var order models.Order
	err := json.NewDecoder(r.Body).Decode(&order)
	if err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}
	defer r.Body.Close()

	// Add support for priority field
	if r.URL.Query().Get("priority") != "" {
		priority, _ := strconv.Atoi(r.URL.Query().Get("priority"))
		order.Priority = priority
	}

	// Continue with the existing code for creating the order
	// ...
}
