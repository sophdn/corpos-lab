```python
import sys
from urllib.parse import urlparse

def validate_url_field(url):
    # Check if the URL field is present and non-empty
    if not url:
        print("Error: URL field is empty.")
        sys.exit(1)

    # Parse the URL
    parsed_url = urlparse(url)

    # Validate the URL components
    if not parsed_url.scheme or not parsed_url.netloc:
        print("Error: Malformed URL.")
        sys.exit(1)

    # Optionally, validate the scheme to be either http or https
    if parsed_url.scheme not in ['http', 'https']:
        print("Error: URL scheme must be 'http' or 'https'.")
        sys.exit(1)

    # If all checks pass, the URL is valid
    return True

# Example usage within the validator script
def main():
    # Assuming 'auth_service_url' is fetched from config
    auth_service_url = "http://example.com"  # Replace with actual config fetching logic
    validate_url_field(auth_service_url)
    # Continue with the rest of the validation logic

if __name__ == "__main__":
    main()
```
