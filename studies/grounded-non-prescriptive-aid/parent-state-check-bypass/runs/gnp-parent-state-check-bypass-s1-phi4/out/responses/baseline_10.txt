import sys
from urllib.parse import urlparse

def validate_url_field(url):
    if not url:
        print("URL field is empty.")
        return 1

    parsed_url = urlparse(url)
    if not parsed_url.scheme or not parsed_url.netloc:
        print("Malformed URL: missing scheme or domain.")
        return 1

    # Additional checks can be added here if necessary
    # e.g., check for valid port, path, etc.

    return 0

# Example usage in the main function
def main():
    auth_service_url = "http://example.com"  # This would be dynamically set
    exit_code = validate_url_field(auth_service_url)
    sys.exit(exit_code)

if __name__ == "__main__":
    main()
