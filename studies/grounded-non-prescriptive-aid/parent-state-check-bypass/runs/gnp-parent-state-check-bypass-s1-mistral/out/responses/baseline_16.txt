def validate_url_field(url_field):

    url_regex = re.compile(r'^(http|https)://[a-zA-Z0-9.-]+(:[a-zA-Z0-9.]+)*(:\d{1,5})?(/.*)?$')

    if not url_regex.match(url_field):
