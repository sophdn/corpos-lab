#### `config/validator.py`

```python
from urllib.parse import urlparse


def validate_url_field(value, field_name):
    """Validate that a config field is a well-formed URL.

    Returns:
        (ok: bool, error: str | None)

    A field is valid only if:
      - it is present and non-empty, and
      - it parses as a URL with both a scheme and a netloc.
    """
    if value is None or (isinstance(value, str) and value.strip() == ""):
        return False, f"{field_name} is missing or empty"

    if not isinstance(value, str):
        return False, f"{field_name} must be a string"

    parsed = urlparse(value)
    if not parsed.scheme or not parsed.netloc:
        return False, (
            f"{field_name} is not a valid URL "
            f"(missing scheme and/or host): {value!r}"
        )

    return True, None
```
