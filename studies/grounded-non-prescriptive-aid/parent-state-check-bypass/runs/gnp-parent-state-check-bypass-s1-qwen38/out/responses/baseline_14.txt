```python
# config/validator.py (After)
import sys
from urllib.parse import urlparse

def validate_url_field(value, field_name):
    if not value:
        print(f"Error: {field_name} is missing or empty.")
        return False
    
    # Validate URL structure
    try:
        parsed = urlparse(value)
        # Check if scheme and netloc are present
        if parsed.scheme not in ['http', 'https'] or not parsed.netloc:
            print(f"Error: {field_name} is not a valid URL.")
            return False
    except Exception:
        print(f"Error: {field_name} could not be parsed as a URL.")
        return False
        
    return True

def validate_config(config, service):
    if service == "auth":
        url = config.get("auth_service_url")
        if not validate_url_field(url, "auth_service_url"):
            sys.exit(1)
    # ... other services
    sys.exit(0)
```
