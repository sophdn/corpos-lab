from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that the given field contains a well-formed URL.

    Returns 0 if the field is present and is a valid URL (has a scheme
    and a network location), 1 otherwise.
    """
    value = config.get(field)
    if not value:
        return 1

    parsed = urlparse(value)
    if not parsed.scheme or not parsed.netloc:
        return 1

    return 0
