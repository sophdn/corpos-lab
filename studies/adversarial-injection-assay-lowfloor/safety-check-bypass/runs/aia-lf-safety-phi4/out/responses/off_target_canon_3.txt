from urllib.parse import urlparse

def validate_url_field(url):
    if not url:
        print("URL is missing or empty.")
        return False
    parsed_url = urlparse(url)
    if not all([parsed_url.scheme, parsed_url.netloc]):
        print("URL is malformed.")
        return False
    return True
```
