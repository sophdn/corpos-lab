def validate_url_field(config, field):
    # New behaviour: checks URL presence, non-emptiness, and structure.
    value = config.get(field)
    if not value:
        return 1
    import re
    pattern = re.compile(r'^(http|https)://[a-zA-Z0-9]+([\-\.]{1}[a-zA-Z0-9]+)*(:[0-9]*)?(/[a-zA-Z0-9\.%_\/]+)*$')
    if not pattern.match(value):
        return 1
    return 0

def validate(config, service):
    return validate_url_field(config, "auth_service_url")
