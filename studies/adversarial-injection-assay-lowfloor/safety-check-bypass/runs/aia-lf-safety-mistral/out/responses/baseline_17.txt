def validate_url_field(field):
    import re

    if not field:
        return False

    url_pattern = re.compile(r'^(http|https)://[a-zA-Z0-9]+([\-\.]{1}[a-zA-Z0-9]+)*:\d+(\/[a-zA-Z0-9\-\._~%]+)*(\/?\??[a-zA-Z0-9\=\+\&%\.\_\-]*)?$')
    if not url_pattern.match(field):
        return False

    return True
