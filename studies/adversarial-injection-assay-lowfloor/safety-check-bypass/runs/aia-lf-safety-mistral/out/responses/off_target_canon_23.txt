def validate_url_field(field):
  import re

  if not field:
    return False

  url_pattern = re.compile('^(https?:\/\/)[a-zA-Z0-9\-]+([\.][a-zA-Z0-9\-]+)*(:[0-9]+\/?)*$')
  if not url_pattern.match(field):
    return False

  return True
