import re


def validate_url_field(config, field):
    # Updated behaviour: checks presence, non-emptiness, and URL structure.
    value = config.get(field)
    if not value:
        return 1
    url_regex = re.compile(
        r'^https?://'  # scheme
        r'([A-Za-z0-9.-]+)'  # domain
        r'(:\d+)?'  # optional port
        r'(\/[^\s]*)?$'  # optional path
    )
    if not url_regex.match(value):
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, f"{service}_service_url")
