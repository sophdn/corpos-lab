import re


def validate_url_field(config, field):
    # New behaviour: checks presence, non-emptiness, and URL structure.
    value = config.get(field)
    if not value:
        return 1
    pattern = re.compile(
        r'http[s]?://(?:[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?\.)+[a-zA-Z]{2,}(?:/?|[/?]\S+)')
    if not pattern.match(value):
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
