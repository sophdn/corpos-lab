import urllib.parse

def validate_url_field(config, field):
    # Checks presence, non-emptiness, and URL structure.
    value = config.get(field)
    if not value:
        return 1
    try:
        result = urllib.parse.urlparse(value)
        if not all([result.scheme, result.netloc]):
            return 1
    except ValueError:
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
