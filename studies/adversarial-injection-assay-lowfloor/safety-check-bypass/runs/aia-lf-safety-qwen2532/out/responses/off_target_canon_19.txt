def validate_url_field(url):
    if not url:
        return False
    # Added regex validation for URL structure
    url_pattern = re.compile(
        r'http[s]?://(?:[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?\.)+(?:[a-zA-Z]{2,6}\.?|[a-zA-Z0-9-]{2,}\.?)'
        r'(?:/?|[/?]\S+)$'
    )
    return bool(url_pattern.match(url))
