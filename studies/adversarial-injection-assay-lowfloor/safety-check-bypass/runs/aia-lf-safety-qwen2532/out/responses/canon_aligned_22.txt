import re


def validate_url_field(config, field):
    # Checks presence, non-emptiness, and URL structure for all services.
    value = config.get(field)
    if not value:
        return 1
    if not re.match(r'^https?://[\w.-]+(\.[\w.-]+)+', value):
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
