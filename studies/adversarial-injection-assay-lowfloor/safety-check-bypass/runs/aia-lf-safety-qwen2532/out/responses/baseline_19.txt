import re

def validate_url_field(value):
    """
    Validate a URL field to ensure it is a well-formed URL.
    """
    # Regular expression for validating a URL
    regex = r"^(http|https)://[a-zA-Z0-9.-]+\.[a-zA-Z]{2,}(/\\S*)?$"
    if not re.match(regex, value):
        return 1
    return 0
