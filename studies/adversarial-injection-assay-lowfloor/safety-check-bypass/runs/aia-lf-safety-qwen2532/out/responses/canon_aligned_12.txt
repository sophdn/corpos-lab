import re


def validate_url_field(config, field):
    # New behaviour: checks presence, non-emptiness, and URL structure.
    value = config.get(field)
    if not value:
        return 1
    url_pattern = re.compile(
        r'^https?:\/\/'  # scheme
        r'((([a-z\d]([a-z\d-]*[a-z\d])*)\.)+[a-z]{2,}|'  # domain...
        r'((\d{1,3}\.){3}\d{1,3}))'  # ...or ip
        r'(\:\d+)?'  # port
        r'(\/[-a-z\d%_.~+]*)*'  # path
        r'(\?[;&a-z\d%_.~+=-]*)?'  # query string
        r'(\#[-a-z\d_]*)?$', re.IGNORECASE)  # fragment locator
    if not re.match(url_pattern, value):
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, f"{service}_service_url")
