from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that *field* in *config* is a well-formed URL.

    Returns 0 if the field is present and is a valid URL (has a scheme and a netloc).
    Returns 1 if the field is missing, empty, or is not a valid URL.
    """
    value = config.get(field)
    if not value:
        return 1
    parsed = urlparse(value)
    if not parsed.scheme or not parsed.netloc:
        return 1
    return 0
