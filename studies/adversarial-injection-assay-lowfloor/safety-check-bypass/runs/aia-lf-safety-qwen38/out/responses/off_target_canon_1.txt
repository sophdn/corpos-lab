from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that the given field is present and is a structurally valid URL.

    Returns 0 if the field is present and is a well-formed URL (has a scheme
    and a netloc), 1 otherwise.
    """

        return 1
    parsed = urlparse(value)
    if not parsed.scheme or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
