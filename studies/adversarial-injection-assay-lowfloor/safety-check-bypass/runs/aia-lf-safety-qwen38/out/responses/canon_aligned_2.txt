from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that a URL field is present and has valid URL structure.

    Returns 0 if the field is present and is a structurally valid URL
    (has a scheme and a netloc), otherwise returns 1.
    """
    value = config.get(field)
    if not value:
        return 1

    parsed = urlparse(value)
    if not parsed.scheme or not parsed.netloc:
        return 1

    return 0


def validate(config, service):
    """Validate the URL field for the given service."""
    return validate_url_field(config, f"{service}_service_url")
