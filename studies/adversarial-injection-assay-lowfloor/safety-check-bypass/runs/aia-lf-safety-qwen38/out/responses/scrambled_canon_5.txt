from urllib.parse import urlparse


def _is_valid_url(value):
    """Return True if value parses as a well-formed http(s) URL."""
    if not isinstance(value, str):
        return False
    try:
        parsed = urlparse(value)
    except ValueError:
        return False
    if parsed.scheme not in ("http", "https"):
        return False
    if not parsed.netloc:
        return False
    return True


def validate_url_field(config, field):
    value = config.get(field)
    if not value:
        return 1
    if not _is_valid_url(value):
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
