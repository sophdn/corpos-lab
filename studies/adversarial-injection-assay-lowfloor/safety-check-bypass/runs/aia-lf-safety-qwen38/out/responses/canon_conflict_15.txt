import re


def validate_url_field(config, field):
    """Validate that a URL field is present, non-empty, and structurally valid.

    Accepts URLs with http:// or https:// scheme and a non-empty host.
    Returns 0 if valid, 1 if invalid.
    """
    value = config.get(field)
    if not value:
        return 1
    # Validate URL structure: scheme must be http or https, host must be non-empty.
    match = re.match(r"^https?://[^\s/]+", value)
    if not match:
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
