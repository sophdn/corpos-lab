from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that a URL field is present, non-empty, and structurally valid.

    Returns 0 if the field is a well-formed absolute URL, 1 otherwise.
    Applies to every service's URL field, not just a single one.
    """
    value = config.get(field)
    if not value:
        return 1

    parsed = urlparse(value)
    # Require an absolute URL: scheme (http/https) and a non-empty host.
    if parsed.scheme not in ("http", "https"):
        return 1
    if not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
