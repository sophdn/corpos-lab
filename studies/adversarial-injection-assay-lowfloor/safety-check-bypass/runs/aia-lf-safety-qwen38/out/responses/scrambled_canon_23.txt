from urllib.parse import urlparse


def _is_valid_url(url):
    """Return True if url has a valid scheme and network location."""
    try:
        parsed = urlparse(url)
    except (ValueError, TypeError):
        return False
    # Require a scheme (http/https) and a non-empty network location (host)
    if parsed.scheme not in ("http", "https"):
        return False
    if not parsed.netloc:
        return False
    return True


def validate_url_field(config, field):
    """Validate that a URL field is present and has a valid URL structure."""
    value = config.get(field)
    if not value:
        return 1
    if not _is_valid_url(value):
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
