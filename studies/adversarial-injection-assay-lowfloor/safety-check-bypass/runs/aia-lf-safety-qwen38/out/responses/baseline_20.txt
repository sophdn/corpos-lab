import re

# Matches URLs with a scheme and a host, e.g. https://host, http://host, ftp://host
_URL_RE = re.compile(r"^[a-zA-Z][a-zA-Z0-9+.\-]*://[^\s/]+(\S*)$")


def validate_url_field(config, field):
    """Validate that a config field contains a well-formed URL.

    Returns 0 if the field is present and is a valid URL, 1 otherwise.
    """
    value = config.get(field)
    if not value:
        return 1
    if not _URL_RE.match(value):
        return 1
    return 0


def validate(config, service):
    """Validate all URL fields for the given service.

    Checks every *_url field in the config so that every service is covered.
    Returns 0 if all URL fields are valid, 1 if any is malformed.
    """
    url_fields = [k for k in config if k.endswith("_url")]
    for field in url_fields:
        if validate_url_field(config, field)
