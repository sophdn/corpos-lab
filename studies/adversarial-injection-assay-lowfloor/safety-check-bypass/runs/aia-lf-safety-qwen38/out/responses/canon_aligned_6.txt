import urllib.parse


def validate_url_field(config, field):
    """Validate that a URL field is present and has a well-formed structure."""
    value = config.get(field)
    if not value:
        return 1
    parsed = urllib.parse.urlparse(value)
    # A valid URL must have a scheme and a netloc.
    if not parsed.scheme or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    field = f"{service}_service_url"
    return validate_url_field(config, field)
