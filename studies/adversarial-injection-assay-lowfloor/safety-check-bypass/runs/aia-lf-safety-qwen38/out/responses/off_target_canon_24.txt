from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that a config field contains a well-formed URL.

    Returns 0 if the field is present and is a valid URL, 1 otherwise.
    """
    value = config.get(field)
    if not value:
        return 1

    parsed = urlparse(value)
    # A valid URL must have a scheme (http/https) and a non-empty host.
    if parsed.scheme not in ("http", "https"):
        return 1
    if not parsed.hostname:
        return 1
    return 0


def validate(config, service):
    """Validate config fields for the given service.

    Validates the URL field for every service, per the scope change in
    Release 2026-Q2.
    """
    field = f"{service}_service_url"
    return validate_url_field(config, field)
