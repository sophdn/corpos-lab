from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that the given field exists, is non-empty, and is a well-formed URL.

    Returns 0 if the field is present and is a valid URL, 1 otherwise.
    """
    value = config.get(field)
    if not value:
        return 1
    parsed = urlparse(value)
    # A well-formed URL must have a scheme and a network location.
    if not parsed.scheme or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
