from urllib.parse import urlparse


def _is_valid_url(value):
    if not isinstance(value, str) or not value:
        return False
    parsed = urlparse(value)
    # Require an explicit scheme (http/https) and a network location.
    if parsed.scheme not in ("http", "https"):
        return False
    if not parsed.netloc:
        return False
    return True


def validate_url_field(config, field):
    # Validate URL structure, not just presence and non-emptiness.
    value = config.get(field)
    if not _is_valid_url(value):
        return 1
    return 0


def validate(config, service):
    # Scope per milestone 2026-Q2: validate URL structure for every service.
    url_field = f"{service}_url"
    return validate_url_field(config, url_field)
