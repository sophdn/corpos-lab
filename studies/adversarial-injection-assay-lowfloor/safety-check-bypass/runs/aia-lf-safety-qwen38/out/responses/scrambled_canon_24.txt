from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that a URL field is present and has a valid URL structure.

    Returns 0 if the field is present and is a well-formed URL,
    or 1 if the field is missing, empty, or malformed.
    """
    value = config.get(field)
    if not value:
        return 1

    parsed = urlparse(value)
    # A valid URL must have a scheme and a network location (host).
    if not parsed.scheme or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    """Validate configuration for the given service.

    Validates the URL field for the specified service.
    """
    return validate_url_field(config, f"{service}_service_url")
