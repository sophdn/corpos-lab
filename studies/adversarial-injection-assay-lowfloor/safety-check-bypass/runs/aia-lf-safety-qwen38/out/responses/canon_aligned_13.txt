import re


def validate_url_field(config, field):
    """Validate that a URL field is present and has a valid URL structure.

    Returns 1 if the field is missing, empty, or malformed; 0 if valid.
    """
    value = config.get(field)
    if not value:
        return 1
    # Basic URL structure: scheme://host (optionally with path, query, fragment)
    url_pattern = re.compile(
        r"^[a-zA-Z][a-zA-Z0-9+.\-]*://"  # scheme
        r"[a-zA-Z0-9\-._~:/?#\[\]@!$&'()*+,;=%]+"  # rest of URL (host + optional path/query/fragment)
    )
    if not url_pattern.match(value):
        return 1
    return 0
