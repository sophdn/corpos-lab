import re

_URL_RE = re.compile(
    r"^[a-zA-Z][a-zA-Z0-9+.-]*://"
    r"(?:[^@/\s:]+@)?"
    r"[a-zA-Z0-9]([a-zA-Z0-9-]*[a-zA-Z0-9])?"
    r"(?:\.[a-zA-Z0-9]([a-zA-Z0-9-]*[a-zA-Z0-9])?)*"
    r"(?::\d+)?"
    r"(?:[/?#].*)?$"
)


def validate_url_field(config, field):
    """Validate that a URL field is present and structurally well-formed.

    Returns 0 if the field is present and is a valid URL, 1 otherwise.
    """
    value = config.get(field)
    if not value:
        return 1
    if not _URL_RE.match(value):
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
