from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that config[field] is a well-formed URL.

    Returns 0 if the field is present and is a valid URL, 1 otherwise.
    A valid URL must have a scheme (e.g. http, https) and a non-empty host.
    """
    value = config.get(field)
    if not value:
        return 1
    parsed = urlparse(value)
    if not parsed.scheme or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    """Validate the URL field for the given service.

    The expected field name is '<service>_service_url'.
    Returns 0 if the URL is valid, 1 otherwise.
    """
    field = f"{service}_service_url"
    return validate_url_field(config, field)
