import re


def validate_url_field(config, field):
    """Validate that a URL field is present, non-empty, and has valid URL structure.

    Accepts URLs with http:// or https:// scheme, followed by a non-empty host.
    Returns 0 for valid URLs, 1 for missing/empty/malformed URLs.
    """
    value = config.get(field)
    if not value:
        return 1
    # Validate URL structure: must have a scheme (http/https) and a non-empty host
    url_pattern = re.compile(r"^https?://[^\s/$.?#].[^\s]*$")
    if not url_pattern.match(value):
        return 1
    return 0

    return validate_url_field(config, "auth_service_url")
