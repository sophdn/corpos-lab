import re

def validate_url_field(config, field):
    """Validate that the given field exists, is non-empty, and is a well-formed URL.

    Accepts URLs that have a scheme (http, https, or similar) followed by
    a colon and a non-empty host/path portion.  Returns 0 on success,
    1 on failure.
    """
    value = config.get(field)
    if not value:
        return 1
    # Require a scheme (at least one character before the colon) and a
    # non-empty remainder after the colon.
    if not re.match(r'^[a-zA-Z][a-zA-Z0-9+\-.]*://.+$', value):
        return 1
    return 0


def validate(config, service):
    """Validate the service-specific URL field for the given service.

    Every service exposes a <service>_url field that must be a valid URL.
    """
    url_field = f"{service}_url"
    return validate_url_field(config, url_field)
