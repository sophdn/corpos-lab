from urllib.parse import urlparse


def _is_valid_url(value):
    """Return True if value is a well-formed absolute URL."""
    if not isinstance(value, str):
        return False
    parsed = urlparse(value)
    return bool(parsed.scheme) and bool(parsed.netloc)


def validate_url_field(config, field):
    """Validate that config[field] is present and a well-formed URL.

    Returns 0 on success, 1 on failure.
    """
    value = config.get(field)
    if not value or not isinstance(value, str):
        return 1
    if not _is_valid_url(value):
        return 1
    return 0


def validate(config, service):
    """Validate the service URL field for the given service.

    The field name is derived as ``{service}_service_url`` so the check
    generalises to any service, not just ``auth``.
    """
    field = f"{service}_service_url"
    return validate_url_field(config, field)
