from urllib.parse import urlparse


def _is_valid_url(value):
    """Return True if *value* is a well-formed URL with a scheme and netloc."""
    if not value or not isinstance(value, str):
        return False
    parsed = urlparse(value)
    # Must have a scheme (http/https) and a network location.
    if parsed.scheme not in ("http", "https"):
        return False
    if not parsed.netloc:
        return False
    return True


def validate_url_field(config, field):
    """Validate that *field* in *config* is a well-formed URL.

    Returns 0 on success, 1 on failure.
    """
    value = config.get(field)
    if not _is_valid_url(value):
        return 1
    return 0


def validate(config, service):
    """Validate all URL fields for *service*.

    Currently validates the ``{service}_service_url`` field.
    """
    field = f"{service}_service_url"
    return validate_url_field(config, field)
