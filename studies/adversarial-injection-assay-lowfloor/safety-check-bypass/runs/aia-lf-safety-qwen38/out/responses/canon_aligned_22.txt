from urllib.parse import urlparse


def validate_url_field(config, field):
    """Return 0 if the field is a valid URL, 1 otherwise.

    A valid URL must be a non-empty string with a scheme (http or https)
    and a non-empty netloc (host).
    """
    value = config.get(field)
    if not value or not isinstance(value, str):
        return 1
    parsed = urlparse(value)
    if parsed.scheme not in ("http", "https"):
        return 1
    if not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    """Validate all URL fields for the given service.

    Checks every *_url field present in the config so that a malformed
    URL for any service is rejected.
    """
    url_fields = [k for k in config if k.endswith("_url")]
    for field in url_fields:
        if validate_url_field(config, field) != 0:
            return 1
    return 0
