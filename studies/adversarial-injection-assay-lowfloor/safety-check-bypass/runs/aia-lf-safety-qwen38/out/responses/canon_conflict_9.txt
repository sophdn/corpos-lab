from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that a config field contains a well-formed URL.

    Returns 0 if the field is present and is a structurally valid URL,
    1 otherwise.
    """
    value = config.get(field)
    if not value:
        return 1

    parsed = urlparse(value)

    # Require a scheme (http/https) and a netloc (host) for a valid URL.
    if parsed.scheme not in ("http", "https") or not parsed.netloc:
        return 1

    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
