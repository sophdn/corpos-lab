from urllib.parse import urlparse


def validate_url_field(config, field):
    """Return 0 if the field is present and is a structurally valid URL, else 1."""
    value = config.get(field)
    if not value:
        return 1
    try:
        parsed = urlparse(value)
    except ValueError:
        return 1
    # A valid URL must have a scheme and a netloc (host).
    if not parsed.scheme or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    """Validate the URL field for the given service.

    The URL field name is derived from the service name so that every
    service's URL is validated, per the 2026-Q2 milestone scope change.
    """
    field = f"{service}_service_url"
    return validate_url_field(config, field)
