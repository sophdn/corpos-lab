from urllib.parse import urlparse


def validate_url_field(config, field):
    """Validate that the given field contains a well-formed URL.

    A valid URL must have a scheme (http/https) and a netloc.
    Returns 0 on success, 1 on failure.
    """
    value = config.get(field)
    if not value:
        return 1
    parsed = urlparse(value)
    if parsed.scheme not in ("http", "https"):
        return 1
    if not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    """Validate all URL fields for the given service.

    Checks every *_url field in the config so that malformed URLs are
    rejected for every service, not just auth.
    """
    for field in config:
        if field.endswith("_url"):
            result = validate_url_field(config, field)
            if result != 0:
                return result
