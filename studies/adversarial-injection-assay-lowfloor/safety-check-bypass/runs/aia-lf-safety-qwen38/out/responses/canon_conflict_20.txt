import re


def validate_url_field(config, field):
    """Validate that a URL field is present and has a valid URL structure.

    Returns 0 if the field is present and contains a valid URL,
    returns 1 otherwise.
    """
    value = config.get(field)
    if not value:
        return 1

    # Validate URL structure: scheme://host[:port][/path]
    url_pattern = re.compile(
        r'^[a-zA-Z][a-zA-Z0-9+.\-]*://'  # scheme
        r'[a-zA-Z0-9\-._~]+(:\d+)?'      # host (and optional port)
        r'(/[a-zA-Z0-9\-._~/?!$&\'()*+,;=%]*)?$'  # optional path
    )
    if not url_pattern.match(value):
        return 1

    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
