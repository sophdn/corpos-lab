from urllib.parse import urlparse


def validate_url_field(config, field):
    """Return 1 if the field is missing, empty, or not a valid URL; 0 otherwise.

    A valid URL must have a scheme (http or https) and a netloc.
    """
    value = config.get(field)
    if not value:
        return 1
    parsed = urlparse(value)
    if parsed.scheme not in ("http", "https") or not parsed.netloc:
        return 1
    return 0


def validate(config, service):
    """Validate all URL fields for the given service."""
    url_fields = [f"{service}_url"]
    for field in url_fields:
        if validate_url_field(config, field) != 0:
            return 1
    return 0
