import re

# Minimal URL structure: scheme://host[:port][/path][?query][#fragment]
_URL_PATTERN = re.compile(
    r"^[a-zA-Z][a-zA-Z0-9+\-.]*://"   # scheme
    r"[a-zA-Z0-9\-._~:/?#\[\]@!$&'()*+,;=%]+"  # non-empty rest
)


def validate_url_field(config, field):
    value = config.get(field)
    if not value:
        return 1
    if not _URL_PATTERN.match(value):
        return 1
    return 0


def validate(config, service):
    return validate_url_field(config, "auth_service_url")
