```python
import re

def validate_url_field(field):
    url_pattern = re.compile(r'http[s]?://(?:[a-zA-Z]|[0-9]|[$-_@.&+]|[!*\\(\\),]|(?:%[0-9a-fA-F][0-9a-fA-F]))+')

    if not url_pattern.match(field):
        print("Malformed URL detected: ", field)
        return 1

    if not field:
        print("The field is empty.")
        return 1

    return 0
```
