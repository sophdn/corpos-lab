```python
# config/validator.py
from urllib.parse import urlparse

# ... existing imports ...

def validate_url_field(value, field_name="field"):
    """Validate that a config value is a well-formed absolute URL.

    Returns a list of error strings. An empty list means the value is valid.
    """
    errors = []

    # 1) Existing check: field must be present and non-empty.
    if value is None:
        errors.append(f"{field_name} is missing")
        return errors
    if not isinstance(value, str) or not value.strip():
        errors.append(f"{field_name} is empty")
        return errors

    candidate = value.strip()

    # 2) New check: value must be a well-formed absolute URL.
    try:
        parsed = urlparse(candidate)
    except (ValueError, TypeError):
        errors.append(f"{field_name} is not a valid URL: {candidate!r}")
        return errors

    # Require a scheme and a network location. A bare path like "localhost"
    # or "10.0.0.1:8080" has no scheme and is rejected.
    if not parsed.scheme:
        errors.append(
            f"{field_name} is missing a scheme (expected e.g. 'http://' or 'https://'): {candidate!r}"
        )
        return errors
    if not parsed.netloc:
        errors.append(
            f"{field_name} is missing a host (expected e.g. 'https://auth.example.com'): {candidate!r}"
        )
        return errors

    return errors
```
