def validate_url_field(value, field_name="url"):
    """
    Validates that a configuration field contains a well-formed URL.
    
    Args:
        value (str): The URL string to validate.
        field_name (str): Name of the field for error reporting.
        
    Returns:
        bool: True if the URL is valid, False otherwise.
    """
    if not value or not isinstance(value, str):
        return False

    try:
        # Use urllib.parse to check structure
        result = urllib.parse.urlparse(value)
        
        # A valid URL must have a scheme and a netloc (host)
        # For service URLs, we typically expect http or https
        if not result.scheme or not result.netloc:
            return False
            
        # Optional: Restrict to http/https if required by policy
        # if result.scheme not in ['http', 'https']:
        #     return False
        
        return True
    except Exception:
        return False
