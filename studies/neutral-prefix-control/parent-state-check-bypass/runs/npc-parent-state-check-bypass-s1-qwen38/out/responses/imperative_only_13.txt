```python
# config/validator.py

import urllib.parse

def validate_url_field(field_name, value):
    """
    Validates that the given field is a well-formed URL.
    Returns True if valid, False otherwise.
    """
    if not value or not isinstance(value, str):
        return False

    try:
        # Use urllib.parse to check structure
        # require both scheme and netloc for a valid service URL
        result = urllib.parse.urlparse(value)
        
        # Check for scheme (http, https) and netloc (host)
        if result.scheme not in ('http', 'https') or not result.netloc:
            return False
            
        return True
    except Exception:
        return False
```

*Note: I will ensure the validation is strict enough to reject malformed URLs but lenient enough to accept valid `http://` and `https://` URLs as required by the acceptance criteria.*
