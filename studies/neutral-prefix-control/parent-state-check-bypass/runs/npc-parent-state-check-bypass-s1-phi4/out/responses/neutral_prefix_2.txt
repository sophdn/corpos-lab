from urllib.parse import urlparse

def validate_url_field(url):
    # Check if the URL is present and non-empty
    if not url:
        print("Error: URL is empty or missing.")
        return False

    # Parse the URL to validate its structure
    parsed_url = urlparse(url)

    # Check if the URL has a valid scheme and netloc
    if not parsed_url.scheme or not parsed_url.netloc:
        print("Error: Malformed URL.")
        return False

    # Additional checks can be added here if needed
    # For example, checking for specific schemes or domains

    # If all checks pass, the URL is valid
    return True

# Example usage in the validator
def config_validator():
    # Example URL for testing
    auth_service_url = "http://example.com"
    
    if not validate_url_field(auth_service_url):
        exit(1)
    else:
        exit(0)
