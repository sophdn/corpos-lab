```python
from urllib.parse import urlparse
import sys

def validate_url_field(url):
    if not url:
        print("Error: URL field is empty.")
        sys.exit(1)
    
    try:
        result = urlparse(url)
        # Check if the URL has a scheme and netloc
        if not all([result.scheme, result.netloc]):
            print("Error: Malformed URL.")
            sys.exit(1)
    except Exception as e:
        print(f"Error: URL validation failed with exception: {e}")
        sys.exit(1)
    
    # If the URL is valid, no exit here
    return True
```
