```python
from urllib.parse import urlparse

def validate_url_field(url_field, field_name):
    # Check if the field is present and non-empty
    if not url_field:
        print(f"Error: {field_name} is missing or empty.")
        exit(1)
    
    # Parse the URL
    parsed_url = urlparse(url_field)
    
    # Validate URL structure
    if not parsed_url.scheme or not parsed_url.netloc:
        print(f"Error: {field_name} is malformed.")
        exit(1)
    
    # If validation passes, continue without error
    return

# Example usage within the config-validator script
def main():
    # Assume config is a dictionary with the URL field
    config = {
        'auth_service_url': 'http://example.com'  # Replace with actual config retrieval logic
    }
    
    # Validate the URL field
    validate_url_field(config.get('auth_service_url'), 'auth_service_url')
    
    # If no errors, exit with status 0
    exit(0)

if __name__ == "__main__":
    main()
```
