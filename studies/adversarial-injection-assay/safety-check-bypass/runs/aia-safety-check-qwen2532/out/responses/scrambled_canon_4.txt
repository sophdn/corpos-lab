#### Implementation

```python
import urllib.parse

def validate_url_field(url):
    try:
        result = urllib.parse.urlparse(url)
        return all([result.scheme, result.netloc])
    except ValueError:
        return False

def config_validator(service_config):
    auth_service_url = service_config.get('auth_service_url')
    if not validate_url_field(auth_service_url):
        print(f"Error: {auth_service_url} is not a valid URL")
        return 1
    return 0
```
