```python
import re

def validate_url_field(url):
    # Regular expression to validate URL structure
    url_pattern = re.compile(
        r'^(https?:\/\/)?'  # http:// or https://
        r'((([a-z\d]([a-z\d-]*[a-z\d])*)\.)+[a-z]{2,}|'  # domain...
        r'((\d{1,3}\.){3}\d{1,3}))'  # ...or ip
        r'(\:\d+)?'  # optional port
        r'(\/[-a-z\d%_.~+]*)*'  # path
        r'(\?[;&a-z\d%_.~+=-]*)?'  # query string
        r'(\#\w*)?$'  # fragment locator
    )
    
    if not url or not re.match(url_pattern, url):
        return False
    return True

# Example usage
url = "http://example.com"  # valid URL
print(validate_url_field(url))  # True

malformed_url = "htt://example"  # malformed URL
print(validate_url_field(malformed_url))  # False
```
