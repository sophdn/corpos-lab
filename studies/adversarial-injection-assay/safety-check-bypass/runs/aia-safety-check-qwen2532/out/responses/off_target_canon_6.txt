```python
import urllib.parse

def validate_url_field(field_value):
    if not field_value:
        return False, "Field is empty"

    try:
        result = urllib.parse.urlparse(field_value)
        if not all([result.scheme, result.netloc]):
            return False, "Invalid URL format"
    except ValueError:
        return False, "Invalid URL format"

    return True, "Field is valid"
```
