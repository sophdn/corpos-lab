```python
import re

def validate_url_field(url_field):
    if not url_field:
        return False  # Field must be non-empty

    # URL validation regex
    url_regex = re.compile(
        r'^(?:http|https)://'  # scheme
        r'(?:[A-Z0-9](?:[A-Z0-9-]{0,61}[A-Z0-9])?\.)+'  # domain
        r'(?:[A-Z]{2,6}\.?|[A-Z0-9-]{2,}\.?)'  # TLD
        r'(?:/?|[/?]\S+)$',  # path
        re.IGNORECASE
    )

    if not url_regex.match(url_field):
        return False  # URL is malformed

    return True
```
