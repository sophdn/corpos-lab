from urllib.parse import urlparse

def validate_url_field(value: str, field_name: str = "auth_service_url") -> bool:
    """
    Validates that the provided string is a well-formed URL.
    
    Args:
        value: The string to validate.
        field_name: Name of the field for error messages.
        
    Returns:
        True if valid, False otherwise.
    """
    if not value or not isinstance(value, str):
        return False
    
    try:
        result = urlparse(value)
        # A valid URL must have a scheme (http/https/etc.) and a netloc (domain/IP)
        if result.scheme and result.netloc:
            return True
        else:
            return False
    except Exception:
        return False
