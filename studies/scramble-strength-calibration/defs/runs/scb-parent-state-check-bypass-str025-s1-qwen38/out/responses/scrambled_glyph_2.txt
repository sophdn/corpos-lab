import urllib.parse

def validate_url_field(url_string):
    """
    Validates that a URL string is structurally valid.
    Returns True if valid, False otherwise.
    """
    if not url_string or not isinstance(url_string, str):
        return False
    
    try:
        result = urllib.parse.urlparse(url_string)
        # A valid URL must have a scheme and a netloc (host)
        if result.scheme not in ('http', 'https'):
            return False
        if not result.netloc:
            return False
        return True
    except Exception:
        return False
