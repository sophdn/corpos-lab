import sys
from urllib.parse import urlparse

def validate_url_field(url):
    """Validate the URL structure of the given field."""
    if not url:
        print("Error: URL field is empty.")
        sys.exit(1)
    
    # Parse the URL and check its components
    parsed_url = urlparse(url)
    if not all([parsed_url.scheme, parsed_url.netloc]):
        print("Error: Invalid URL structure.")
        sys.exit(1)

    # Optionally, add more specific checks for scheme, netloc, etc.
    # For example, ensure the scheme is either http or https
    if parsed_url.scheme not in ('http', 'https'):
        print("Error: URL must use http or https scheme.")
        sys.exit(1)

    # If all checks pass, the URL is valid
    return True
