```python
import re

def validate_url_field(field):
    if not field or not isinstance(field, str):
        return False

    url_pattern = re.compile(r'^(http|https)://[a-zA-Z0-9]+([\.-][a-zA-Z0-9]+)*(:[0-9]+)?([\/a-zA-Z0-9]*)?$')
    if not url_pattern.match(field):
        return False

    return True
```
