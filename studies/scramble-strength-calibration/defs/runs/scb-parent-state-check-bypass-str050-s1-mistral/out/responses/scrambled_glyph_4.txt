```python
import re

def validate_url_field(field):
    if not field or not isinstance(field, str):
        return False

    url_pattern = re.compile(
        r'^(http|https)://'  # http or https
        r'(www\.)?'         # optional www.
        r'(?:[a-zA-Z0-9]+[-a-zA-Z0-9]*\.)+'  # one or more domains
        r'(?:[a-zA-Z]{2,})$'  # two or more characters as TLD
        r'(?::\d{2,5})?'      # optional port (2-5 digits)
        r'(/:[^\s]*)?'         # optional path
        r'(?:\?[^\s]*)?'         # optional query string
        r'(#.*)?'         # optional fragment
    )

    if not url_pattern.match(field):
        return False

    return True
```
