#### 1. Update `config/validator.py`

Replace the existing `validate_url_field()` logic with proper URL validation using Python’s built-in `urllib.parse` module (no new dependencies required).

```python
# config/validator.py

import sys
from urllib.parse import urlparse

def validate_url_field(config: dict, field_name: str) -> bool:
    """
    Validates that the given field exists, is non-empty, and is a well-formed URL.
    Returns True if valid, False otherwise.
    """
    if field_name not in config:
        print(f"ERROR: Missing required field: {field_name}", file=sys.stderr)
        return False

    value = config[field_name]

    if not isinstance(value, str) or not value.strip():
        print(f"ERROR: Field '{field_name}' is empty or not a string.", file=sys.stderr)
        return False

    value = value.strip()

    # Parse the URL
    try:
        parsed = urlparse(value)
    except Exception as e:
        print(f"ERROR: Field '{field_name}' has invalid URL: {e}", file=sys.stderr)
        return False

    # Require scheme and netloc (host)
    if not parsed.scheme:
        print(f"ERROR: Field '{field_name}' is missing a scheme (e.g., http, https).", file=sys.stderr)
        return False

    if not parsed.netloc:
        print(f"ERROR: Field '{field_name}' is missing a host.", file=sys.stderr)
        return False

    # Optionally: restrict allowed schemes for security (e.g., only http/https)
    allowed_schemes = ("http", "https")
    if parsed.scheme.lower() not in allowed_schemes:
        print(
            f"ERROR: Field '{field_name}' uses disallowed scheme '{parsed.scheme}'. "
            f"Allowed schemes: {', '.join(allowed_schemes)}.",
            file=sys.stderr
        )
        return False

    # Optional: validate port if present
    if parsed.port is not None:
        if not (1 <= parsed.port <= 65535):
            print(
                f"ERROR: Field '{field_name}' has an invalid port {parsed.port}.",
                file=sys.stderr
            )
            return False

    return True
```
