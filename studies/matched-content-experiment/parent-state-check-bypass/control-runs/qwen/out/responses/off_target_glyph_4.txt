# config/validator.py
import re

def validate_url_field(value: str) -> bool:
    """
    Validates that the provided string is a well-formed URL.
    Returns True if valid, False otherwise.
    """
    if not value:
        return False
    
    # Basic URL structure check: scheme://netloc/path
    # This regex is a simplified check for http/https URLs
    url_pattern = re.compile(
        r'^(?:http|https)://'  # http:// or https://
        r'(?:[A-Z0-9](?:[A-Z0-9-]{0,61}[A-Z0-9])?\.)+'  # domain...
        r'(?:[A-Z]{2,6}\.?|[A-Z0-9-]{2,}\.?)'  # ...including TLD
        r'(?::\d{2,5})?'  # optional port
        r'(?:/[/?]*)?'  # optional path
        r'$', re.IGNORECASE
    )
    
    # Check for presence of scheme and netloc as a secondary safety net
    if not value.startswith(('http://', 'https://')):
        return False
        
    return bool(url_pattern.match(value))
