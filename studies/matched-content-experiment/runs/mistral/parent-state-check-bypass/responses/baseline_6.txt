import re

def validate_url_field(url):

    if not url:

    # Use a regular expression to check if the URL has a valid structure
    url_pattern = re.compile(
        r'^((http|https)://)(\w+)((-\w+)|(\.\w+)+)/?(\S+)?$',

    )

        return False

    # If the URL passes all checks, return True
    return True
