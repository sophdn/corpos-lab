import urllib.parse

def validate_url_field(url_string):
    """
    Validates that the provided string is a well-formed URL.
    Returns True if valid, False otherwise.
    """
    if not url_string:
        return False
    
    try:
        result = urllib.parse.urlparse(url_string)
        # A valid URL must have a scheme (http, https, etc.) and a netloc (domain/host)
        if not result.scheme or not result.netloc:
            return False
        return True
    except Exception:
        return False
