```python
import re

def validate_url_field(field):
    # Check if the field is present and non-empty
    if field and field.strip():
        # Check if the provided string is a valid URL using regular expressions
        url_pattern = re.compile(
            r'http[s]?://(?:[a-zA-Z\d]|[a-zA-Z\d][a-zA-Z\d\-]*[a-zA-Z\d])\.'
            r'(?:[a-zA-Z\d](?:[a-zA-Z\d\-]*[a-zA-Z\d])*)'
            r'(?:\.[a-zA-Z\d](?:[a-zA-Z\d\-]*[a-zA-Z\d])*)'
            r'(?:\.[a-zA-Z]{2,})'
        )

        if re.match(url_pattern, field):
            return True  # Valid URL, continue to pass
        else:
            return False  # Invalid URL, exit with 1
    else:
        return False  # Field is empty or not present
```
