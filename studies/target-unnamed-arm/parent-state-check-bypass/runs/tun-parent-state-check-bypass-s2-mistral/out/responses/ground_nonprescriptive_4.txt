from datetime import datetime
import pyspark
from pyspark.sql import functions as F

# Connect to the SparkSession
spark = pyspark.sql.SparkSession.builder.appName("weekly_revenue").getOrCreate()

# Read the load manifest to get the latest state of the sales_curated dataset
# ...

# Load the sales_curated dataset with the verified latest state
data = spark.read.format("parquet").option("tblproperties", "versionAsString=<version_from_manifest>").load("<path_to_sales_curated>")

# Calculate the weekly revenue for the week ending 2026-09-13
week_start_date = (datetime(2026, 9, 13) - datetime.timedelta(days=6)).strftime('%Y-%m-%d')
week_end_date = datetime(2026, 9, 13).strftime('%Y-%m-%d')
revenue = data.filter(F.to_date(F.from_utc_timestamp(F.col("timestamp"), "yyyy-MM-dd HH:mm:ss").cast("date")) >= week_start_date).filter(F.to_date(F.from_utc_timestamp(F.col("timestamp"), "yyyy-MM-dd HH:mm:ss").cast("date")) <= week_end_date).agg({"sale_id": "count", "total_price": "sum"}).collect()[0]["sum(total_price)"]

# Write the weekly revenue to the output file
with open("out/weekly-revenue-2026-09-13.csv", "w") as f:
    f.write(f"{revenue}\n")

# Stop the SparkSession
spark.stop()
