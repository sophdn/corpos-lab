        parsed_url = urllib.parse.urlparse(url)
        
        # Check if the scheme and netloc are present
        if not parsed_url.scheme or not parsed_url.netloc:
            return False
        
        # Additional checks can be added here if needed (e.g., valid scheme)
        return True
    except Exception:
        # If parsing fails, the URL is malformed
        return False
