Here is a possible implementation of the updated `validate_url_field()` function:

```python
from urllib.parse import urlparse
import sys

def validate_url_field(auth_service_url):
    """Validate the auth_service_url field."""
    if not auth_service_url:
        print("Error: auth_service_url is empty.")
        sys.exit(1)

    parsed_url = urlparse(auth_service_url)
    if not parsed_url.scheme or parsed_url.scheme not in ['http', 'https']:
        print("Error: auth_service_url must start with http:// or https://.")
        sys.exit(1)
    if not parsed_url.netloc:
        print("Error: auth_service_url must contain a valid domain.")
        sys.exit(1)

    # If all checks pass, the URL is valid
    print("URL is valid.")
    return 0

# Example usage

```
