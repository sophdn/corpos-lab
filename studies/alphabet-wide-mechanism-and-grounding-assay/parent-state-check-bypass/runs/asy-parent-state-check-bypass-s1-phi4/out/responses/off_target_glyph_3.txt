```python
import urllib.parse

def validate_url_field(url):
    # Check if the URL is present and non-empty
    if not url:
        print("URL field is missing or empty.")
        exit(1)
    
    # Validate URL structure
    try:
        result = urllib.parse.urlparse(url)
        if all([result.scheme, result.netloc]):
            # The URL has a valid scheme and network location
            return
        else:
            raise ValueError("Invalid URL structure.")
    except ValueError as e:
        print(f"Invalid URL: {url}. Error: {e}")
        exit(1)
```
